  <script>
    const DOLLAR_RATE = 5.30;
    const OPERATION_COLORS = [
      '#22c55e','#6366f1','#06b6d4','#f59e0b','#ef4444','#8b5cf6','#14b8a6','#eab308','#3b82f6','#ec4899',
      '#84cc16','#0ea5e9','#f97316','#a855f7','#10b981','#f43f5e','#7c3aed','#0891b2','#65a30d','#2563eb',
      '#dc2626','#d97706','#9333ea','#0d9488','#4f46e5','#db2777','#16a34a','#0284c7','#b45309','#7e22ce'
    ];

    let rawData = [];
    let allData = [];
    let lastUpdatedAt = null;
    const filterState = {
      empresa: [],
      pais: [],
      nicho: []
    };
    let appConfig = {
      dashboard_title: 'Dashboard de Operações'
    };

    function fmt(v) {
      return 'R$ ' + Number(v || 0).toLocaleString('pt-BR', {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2
      });
    }

    function fmtN(v) {
      return Number(v || 0).toLocaleString('pt-BR');
    }

    function fmtUSD(v) {
      return '$' + Number(v || 0).toLocaleString('pt-BR', {
        minimumFractionDigits: 1,
        maximumFractionDigits: 1
      });
    }

    function fmtRoaPct(roas) {
      const pct = (Number(roas || 0) - 1) * 100;
      return pct.toFixed(1).replace('.', ',') + '%';
    }

    function fmtSignedBRL(v) {
      const n = Number(v || 0);
      if (n < 0) return '- ' + fmt(Math.abs(n));
      return fmt(n);
    }

    function fmtDateTime(v) {
      const d = new Date(v);
      return d.toLocaleString('pt-BR');
    }

    function getOps(data) {
      return [...new Set(data.map(d => d.operacao))].sort();
    }

    function getOperationParts(op) {
      const parts = String(op || '').trim().split(/\s+/);
      return {
        empresa: parts[0] || '',
        pais: parts[1] || '',
        nicho: parts[2] || ''
      };
    }

    function getFilterOptions(data) {
      const empresas = new Set();
      const paises = new Set();
      const nichos = new Set();

      data.forEach(item => {
        const p = getOperationParts(item.operacao);
        if (p.empresa) empresas.add(p.empresa);
        if (p.pais) paises.add(p.pais);
        if (p.nicho) nichos.add(p.nicho);
      });

      return {
        empresa: [...empresas].sort(),
        pais: [...paises].sort(),
        nicho: [...nichos].sort()
      };
    }

    function getOperationColorMap(ops) {
      const map = {};
      ops.forEach((op, idx) => {
        map[op] = OPERATION_COLORS[idx % OPERATION_COLORS.length];
      });
      return map;
    }

    function applyFilters() {
      allData = rawData.filter(item => {
        const p = getOperationParts(item.operacao);
        const okEmpresa = filterState.empresa.length === 0 || filterState.empresa.includes(p.empresa);
        const okPais = filterState.pais.length === 0 || filterState.pais.includes(p.pais);
        const okNicho = filterState.nicho.length === 0 || filterState.nicho.includes(p.nicho);
        return okEmpresa && okPais && okNicho;
      });

      const hasData = allData.length > 0;
      document.getElementById('empty-state').classList.toggle('hidden', hasData);
      document.getElementById('dashboard').classList.toggle('hidden', !hasData);
      document.getElementById('record-count').textContent = allData.length + ' registros';
      document.getElementById('last-update').textContent = lastUpdatedAt ? 'Atualizado em: ' + fmtDateTime(lastUpdatedAt) : '-';

      if (!hasData) {
        lucide.createIcons();
        return;
      }

      renderSummary();
      renderOverallCharts();
      renderOpsGrid();
      lucide.createIcons();
    }

    function renderFilterDropdown(type, label, options) {
      const selected = filterState[type];
      const selectedText = selected.length > 0 ? selected.join(', ') : 'Todos';

      return `
        <details class="bg-white/5 rounded-lg p-2 relative" data-filter="${type}">
          <summary class="cursor-pointer list-none flex items-center justify-between text-sm text-gray-200">
            <span>${label}</span>
            <span class="text-xs text-indigo-300 truncate max-w-[70%] text-right">${selectedText}</span>
          </summary>
          <div class="absolute left-0 right-0 top-full mt-2 z-40 bg-[#0f1228] border border-white/10 rounded-lg p-2 shadow-xl">
            <input type="text" data-filter-search="${type}" placeholder="Pesquisar..." class="w-full bg-[#11142b] border border-white/10 rounded px-2 py-1 text-xs text-gray-200 mb-2" />
            <div class="max-h-32 overflow-y-auto space-y-1" data-filter-list="${type}">
              ${options.map(opt => `
                <label class="flex items-center gap-2 text-xs text-gray-300">
                  <input type="checkbox" data-filter-checkbox="${type}" value="${opt}" ${selected.includes(opt) ? 'checked' : ''} />
                  <span>${opt}</span>
                </label>
              `).join('')}
            </div>
          </div>
        </details>
      `;
    }

    function renderFilters() {
      const opts = getFilterOptions(rawData);
      const wrap = document.getElementById('filters-wrap');
      wrap.innerHTML = [
        renderFilterDropdown('empresa', 'Empresa', opts.empresa),
        renderFilterDropdown('pais', 'País', opts.pais),
        renderFilterDropdown('nicho', 'Nicho', opts.nicho)
      ].join('');
    }

    function getVal(data, op, dados) {
      const r = data.find(d => d.operacao === op && d.dados === dados);
      return r ? Number(r.valor || 0) : 0;
    }

    function getMetricsForOp(data, op) {
      const aqG = getVal(data, op, 'AQ - Gasto');
      const aqR = getVal(data, op, 'AQ - Receita');
      const reRBUsd = getVal(data, op, 'RE - Receita - B');
      const reRAUsd = getVal(data, op, 'RE - Receita - A');
      const reRPUsd = getVal(data, op, 'RE - Receita - P');
      const reRB = reRBUsd * DOLLAR_RATE;
      const reRA = reRAUsd * DOLLAR_RATE;
      const reRP = reRPUsd * DOLLAR_RATE;
      const reRe = reRB + reRA + reRP;
      const reSB = getVal(data, op, 'RE - Sessões - B');
      const reSA = getVal(data, op, 'RE - Sessões - A');
      const reSP = getVal(data, op, 'RE - Sessões - P');
      const reSes = reSB + reSA + reSP;
      const totalRev = aqR + reRe;
      const roas = aqG > 0 ? totalRev / aqG : 0;

      return {
        aqG, aqR,
        reRBUsd, reRAUsd, reRPUsd,
        reRB, reRA, reRP, reRe,
        reSB, reSA, reSP, reSes,
        totalRev, roas
      };
    }

    function renderSummary() {
      const ops = getOps(allData);
      let totalGasto = 0, totalReceitaAQ = 0, totalReceitaRE = 0, totalSessoes = 0;
      let totalSesB = 0, totalSesA = 0, totalSesP = 0;

      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        totalGasto += m.aqG;
        totalReceitaAQ += m.aqR;
        totalReceitaRE += m.reRe;
        totalSessoes += m.reSes;
        totalSesB += m.reSB;
        totalSesA += m.reSA;
        totalSesP += m.reSP;
      });

      const totalReceitaTotal = totalReceitaAQ + totalReceitaRE;
      const lucroGeral = totalReceitaTotal - totalGasto;
      const roasGeral = totalGasto > 0 ? totalReceitaTotal / totalGasto : 0;

      const cards = [
        { label: 'Gasto Total', value: fmt(totalGasto), icon: 'trending-down', color: 'text-red-400', bg: 'bg-red-500/10' },
        { label: 'Receita Total', value: fmt(totalReceitaTotal), icon: 'trending-up', color: 'text-emerald-400', bg: 'bg-emerald-500/10' },
        { label: 'Lucro Geral', value: fmtSignedBRL(lucroGeral), icon: 'wallet', color: lucroGeral >= 0 ? 'text-emerald-400' : 'text-red-400', bg: 'bg-violet-500/10' },
        { label: 'Sessões', value: fmtN(totalSessoes), icon: 'users', color: 'text-cyan-400', bg: 'bg-cyan-500/10', detail: `<span class="text-cyan-300 font-semibold">B:</span> <span class="font-semibold text-white">${fmtN(totalSesB)}</span>  <span class="text-amber-300 font-semibold">A:</span> <span class="font-semibold text-white">${fmtN(totalSesA)}</span>  <span class="text-indigo-300 font-semibold">P:</span> <span class="font-semibold text-white">${fmtN(totalSesP)}</span>` },
        { label: 'ROAS Geral', value: fmtRoaPct(roasGeral), icon: 'target', color: 'text-amber-400', bg: 'bg-amber-500/10' }
      ];

      document.getElementById('summary-cards').innerHTML = cards.map((cd, i) => `
        <div class="card-glass rounded-xl p-3 sm:p-4 fade-up fade-up-${i+1}">
          <div class="flex items-center gap-2 mb-2">
            <div class="w-7 h-7 rounded-lg ${cd.bg} flex items-center justify-center">
              <i data-lucide="${cd.icon}" class="w-4 h-4 ${cd.color}"></i>
            </div>
            <span class="text-xs text-gray-400">${cd.label}</span>
          </div>
          <div class="text-lg sm:text-xl font-bold text-white">${cd.value}</div>
          ${cd.detail ? `<div class="text-[11px] text-gray-300 mt-1 leading-relaxed">${cd.detail}</div>` : ''}
        </div>
      `).join('');
    }

    function renderOverallCharts() {
      const ops = getOps(allData);
      const opColorMap = getOperationColorMap(ops);
      const opsBySpendDesc = [...ops].sort((a, b) => {
        const ma = getMetricsForOp(allData, a);
        const mb = getMetricsForOp(allData, b);
        return mb.aqG - ma.aqG;
      });

      const spendData = opsBySpendDesc.map((op) => {
        const m = getMetricsForOp(allData, op);
        return { label: op, value: m.aqG, color: opColorMap[op] };
      }).filter(item => item.value > 0);

      const totalSpend = spendData.reduce((acc, cur) => acc + cur.value, 0);
      const stackedSegments = spendData.map(item => {
        const pct = totalSpend > 0 ? (item.value / totalSpend) * 100 : 0;
        return `
          <div class="h-full" style="width:${Math.max(pct, 1)}%;background:${item.color}" title="${item.label} | Gasto: ${fmt(item.value)} | ${pct.toFixed(1)}%"></div>
        `;
      }).join('');

      const tableRows = spendData.map(item => {
        const pct = totalSpend > 0 ? (item.value / totalSpend) * 100 : 0;
        return `
          <div class="grid grid-cols-[10px_1fr_auto_auto] items-center gap-2 text-[10px] py-0.5 border-b border-white/5">
            <span class="w-2.5 h-2.5 rounded-full" style="background:${item.color}"></span>
            <span class="text-gray-300 truncate">${item.label}</span>
            <span class="text-gray-200 whitespace-nowrap">${fmt(item.value)}</span>
            <span class="text-gray-400 whitespace-nowrap">${pct.toFixed(1)}%</span>
          </div>
        `;
      }).join('');

      document.getElementById('chart-revenue').innerHTML = `
        <div class="h-full flex items-center justify-between gap-2">
          <div class="flex-1">
            <div class="text-[10px] text-gray-400 mb-1">Distribuição percentual do gasto</div>
            <div class="h-8 w-full rounded-md overflow-hidden bg-gray-900 border border-white/10 flex">
              ${stackedSegments || '<div class="w-full h-full bg-gray-800"></div>'}
            </div>
          </div>
          <div class="text-right pr-1">
            <div class="text-[10px] text-gray-400">Operações</div>
            <div class="text-2xl font-bold text-indigo-300">${ops.length}</div>
          </div>
        </div>
      `;
      document.getElementById('revenue-legend').innerHTML = `
        <div class="grid grid-cols-[1fr_auto_auto] gap-2 text-[10px] text-gray-500 pb-1 border-b border-white/10">
          <span>Operação</span><span>Gasto</span><span>%</span>
        </div>
        <div class="max-h-[98px] overflow-y-auto pr-1">
          ${tableRows || '<div class="text-[10px] text-gray-500 py-2">Sem gasto registrado.</div>'}
        </div>
      `;

      const lucroData = opsBySpendDesc.map(op => {
        const m = getMetricsForOp(allData, op);
        return { label: op, value: m.totalRev - m.aqG };
      });
      const maxAbsLucro = Math.max(1, ...lucroData.map(i => Math.abs(i.value)));

      document.getElementById('chart-sessions').innerHTML = `
        <div class="h-full overflow-x-auto overflow-y-hidden">
          <div class="h-full min-w-max flex items-end gap-2 pr-2">
          ${lucroData.map((item, idx) => {
            const magnitude = (Math.abs(item.value) / maxAbsLucro) * 42;
            const isPositive = item.value >= 0;
            return `
              <div class="w-[72px] h-full flex flex-col items-center justify-center shrink-0">
                <div class="text-[9px] ${isPositive ? 'text-emerald-300' : 'text-red-300'} whitespace-nowrap mb-1">${fmtSignedBRL(item.value)}</div>
                <div class="relative w-full h-20">
                  <div class="absolute left-0 right-0 top-1/2 border-t border-white/15"></div>
                  <div class="absolute left-1/2 -translate-x-1/2 w-6 ${isPositive ? 'bg-emerald-500' : 'bg-red-500'} rounded-sm"
                    style="${isPositive ? `bottom:50%;height:${Math.max(magnitude, 3)}%` : `top:50%;height:${Math.max(magnitude, 3)}%`}"></div>
                </div>
                <div class="text-[8px] text-gray-500 truncate w-full text-center">${item.label}</div>
              </div>
            `;
          }).join('')}
          </div>
        </div>
      `;
      document.getElementById('lucro-legend').innerHTML = `
        <div class="grid grid-cols-[1fr_auto] gap-2 text-[10px] text-gray-500 pb-1 border-b border-white/10">
          <span>Operação</span><span>Lucro</span>
        </div>
        <div class="max-h-[98px] overflow-y-auto pr-1">
          ${lucroData.map(item => `
            <div class="grid grid-cols-[1fr_auto] gap-2 text-[10px] py-0.5 border-b border-white/5">
              <span class="text-gray-300 truncate">${item.label}</span>
              <span class="${item.value >= 0 ? 'text-emerald-300' : 'text-red-300'} whitespace-nowrap">${fmtSignedBRL(item.value)}</span>
            </div>
          `).join('')}
        </div>
      `;

      const scatterData = opsBySpendDesc.map((op) => {
        const m = getMetricsForOp(allData, op);
        const roasAq = m.aqG > 0 ? (m.aqR / m.aqG) : 0;
        const roasPct = (roasAq - 1) * 100;
        return { label: op, gasto: m.aqG, roasAq: roasAq, roasPct: roasPct, color: opColorMap[op] };
      });

      const maxX = Math.max(1, ...scatterData.map(d => d.gasto));
      const minY = -50;
      const maxY = 25;
      const spanY = Math.max(maxY - minY, 1);
      const plotW = 270;
      const plotH = 110;
      const margin = { l: 26, r: 10, t: 8, b: 22 };
      const innerW = plotW - margin.l - margin.r;
      const innerH = plotH - margin.t - margin.b;

      const points = scatterData.map(d => {
        const x = margin.l + (d.gasto / maxX) * innerW;
        const y = margin.t + innerH - ((d.roasPct - minY) / spanY) * innerH;
        return `
          <circle cx="${x.toFixed(1)}" cy="${y.toFixed(1)}" r="6.5" fill="transparent">
            <title>${d.label} | Gasto: ${fmt(d.gasto)} | ROAS AQ: ${fmtRoaPct(d.roasAq)}</title>
          </circle>
          <circle cx="${x.toFixed(1)}" cy="${y.toFixed(1)}" r="4.5" fill="${d.color}" opacity="0.95">
            <title>${d.label} | Gasto: ${fmt(d.gasto)} | ROAS AQ: ${fmtRoaPct(d.roasAq)}</title>
          </circle>
        `;
      }).join('');

      const zeroY = margin.t + innerH - ((0 - minY) / spanY) * innerH;
      document.getElementById('chart-roas').innerHTML = `
        <svg viewBox="0 0 ${plotW} ${plotH}" class="w-full h-full">
          <rect x="${margin.l}" y="${margin.t}" width="${innerW}" height="${innerH}" fill="transparent"></rect>
          <line x1="${margin.l}" y1="${margin.t}" x2="${margin.l}" y2="${margin.t + innerH}" stroke="rgba(255,255,255,0.2)" stroke-width="1"></line>
          <line x1="${margin.l}" y1="${margin.t + innerH}" x2="${margin.l + innerW}" y2="${margin.t + innerH}" stroke="rgba(255,255,255,0.2)" stroke-width="1"></line>
          <line x1="${margin.l}" y1="${zeroY}" x2="${margin.l + innerW}" y2="${zeroY}" stroke="rgba(255,255,255,0.12)" stroke-dasharray="3 3" stroke-width="1"></line>
          ${points}
          <text x="${margin.l}" y="${margin.t - 1}" fill="#9ca3af" font-size="8">ROAS AQ (Y)</text>
          <text x="${margin.l + innerW - 40}" y="${plotH - 4}" fill="#9ca3af" font-size="8">Gasto (X)</text>
          <text x="1" y="${margin.t + 8}" fill="#9ca3af" font-size="7">25%</text>
          <text x="1" y="${zeroY + 2}" fill="#9ca3af" font-size="7">0%</text>
          <text x="1" y="${margin.t + innerH}" fill="#9ca3af" font-size="7">-50%</text>
        </svg>
      `;
      document.getElementById('roas-legend').innerHTML = `
        <div class="grid grid-cols-[1fr_auto_auto] gap-2 text-[10px] text-gray-500 pb-1 border-b border-white/10">
          <span>Operação</span><span>Gasto</span><span>ROAS AQ</span>

    function renderOpsGrid() {
      const ops = getOps(allData);
      const grid = document.getElementById('ops-grid');

        const totalRec = m.aqR + m.reRe;

        const totalForPie = Math.max(m.aqR + m.reRe, 1);
        const aqPercent = (m.aqR / totalForPie * 100).toFixed(1);
        const rePercent = (m.reRe / totalForPie * 100).toFixed(1);

        const circumference = 2 * Math.PI * pieRadius;
        const aqArc = (aqPercent / 100) * circumference;
        const reArc = (rePercent / 100) * circumference;

          const width = Math.max(percent, 14);
          return `
            <div>

    function renderAll(data, updatedAt) {
      rawData = data || [];
      lastUpdatedAt = updatedAt || null;
      renderFilters();
      applyFilters();
    }

    function loadData(showToast = false) {
      google.script.run
        .withSuccessHandler(function(rawResult) {
          let result = rawResult;
          if (typeof rawResult === 'string') {
            try {
              result = JSON.parse(rawResult);
            } catch (e) {
              console.error('Falha ao parsear syncDataJson:', e, rawResult);
              result = null;
            }
          }

          console.log('[dashboard] resposta syncData:', result);
          if (!result || typeof result !== 'object') {
            console.warn('syncData retornou resultado inválido:', result);
            renderAll([], null);
            return;
          }

          if (result.success === false) {
            console.error('syncData falhou:', result.error || 'Erro desconhecido');
            alert('Erro ao carregar dados: ' + (result.error || 'Falha na sincronização.'));
            renderAll([], result.updatedAt);
            return;
          }

          appConfig = result.config || appConfig;
          document.getElementById('main-title').textContent = appConfig.dashboard_title || 'Dashboard de Operações';
          renderAll(result.data || [], result.updatedAt);

          if (showToast) {
            const toast = document.getElementById('sync-toast');
            toast.classList.remove('hidden');
            setTimeout(() => toast.classList.add('hidden'), 2500);
          }
        })
        .withFailureHandler(function(err) {
          console.error(err);
          alert('Erro ao carregar dados: ' + (err.message || err));
        })
        .syncDataJson();
    }

    document.getElementById('btn-sync').addEventListener('click', function() {
      const btn = document.getElementById('btn-sync');
      btn.disabled = true;
      loadData(true);
      setTimeout(function() {
        btn.disabled = false;
      }, 600);
    });

    document.getElementById('filters-wrap').addEventListener('change', function(e) {
      const type = e.target.getAttribute('data-filter-checkbox');
      if (!type) return;
      const values = [...document.querySelectorAll(`input[data-filter-checkbox="${type}"]:checked`)].map(el => el.value);
      filterState[type] = values;
      renderFilters();
      applyFilters();
    });

    document.getElementById('filters-wrap').addEventListener('input', function(e) {
      const type = e.target.getAttribute('data-filter-search');
      if (!type) return;
      const term = String(e.target.value || '').toLowerCase();
      const list = e.target.closest('[data-filter]').querySelector(`[data-filter-list="${type}"]`);
      if (!list) return;
      [...list.querySelectorAll('label')].forEach(label => {
        const txt = label.textContent.toLowerCase();
        label.classList.toggle('hidden', !txt.includes(term));
      });
    });

    document.getElementById('btn-clear-filters').addEventListener('click', function() {
      filterState.empresa = [];
      filterState.pais = [];
      filterState.nicho = [];
      renderFilters();
      applyFilters();
    });
