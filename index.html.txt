  <script>
    let allData = [];
    let appConfig = {
      dashboard_title: 'Dashboard de Operações'
    };

    function fmt(v) {
      return 'R$ ' + Number(v || 0).toLocaleString('pt-BR', {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2
      });
    }

    function fmtN(v) {
      return Number(v || 0).toLocaleString('pt-BR');
    }

    function fmtDateTime(v) {
      const d = new Date(v);
      return d.toLocaleString('pt-BR');
    }

    function getOps(data) {
      return [...new Set(data.map(d => d.operacao))].sort();
    }

    function getVal(data, op, dados) {
      const r = data.find(d => d.operacao === op && d.dados === dados);
      return r ? Number(r.valor || 0) : 0;
    }

    function getMetricsForOp(data, op) {
      const aqG = getVal(data, op, 'AQ - Gasto');
      const aqR = getVal(data, op, 'AQ - Receita');
      const reRB = getVal(data, op, 'RE - Receita - B');
      const reRA = getVal(data, op, 'RE - Receita - A');
      const reRP = getVal(data, op, 'RE - Receita - P');
      const reRe = reRB + reRA + reRP;
      const reSB = getVal(data, op, 'RE - Sessões - B');
      const reSA = getVal(data, op, 'RE - Sessões - A');
      const reSP = getVal(data, op, 'RE - Sessões - P');
      const reSes = reSB + reSA + reSP;
      const totalRev = aqR + reRe;
      const roas = aqG > 0 ? totalRev / aqG : 0;

      return { aqG, aqR, reRB, reRA, reRP, reRe, reSB, reSA, reSP, reSes, totalRev, roas };
    }

    function renderSummary() {
      const ops = getOps(allData);
      let totalGasto = 0, totalReceitaAQ = 0, totalReceitaRE = 0, totalSessoes = 0;

      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        totalGasto += m.aqG;
        totalReceitaAQ += m.aqR;
        totalReceitaRE += m.reRe;
        totalSessoes += m.reSes;
      });

      const totalReceitaTotal = totalReceitaAQ + totalReceitaRE;
      const roasGeral = totalGasto > 0 ? totalReceitaTotal / totalGasto : 0;

      const cards = [
        { label: 'Gasto Total', value: fmt(totalGasto), icon: 'trending-down', color: 'text-red-400', bg: 'bg-red-500/10' },
        { label: 'Receita Total', value: fmt(totalReceitaTotal), icon: 'trending-up', color: 'text-emerald-400', bg: 'bg-emerald-500/10' },
        { label: 'Sessões', value: fmtN(totalSessoes), icon: 'users', color: 'text-cyan-400', bg: 'bg-cyan-500/10' },
        { label: 'ROAS Geral', value: roasGeral.toFixed(2) + 'x', icon: 'target', color: 'text-amber-400', bg: 'bg-amber-500/10' }
      ];

      document.getElementById('summary-cards').innerHTML = cards.map((cd, i) => `
        <div class="card-glass rounded-xl p-3 sm:p-4 fade-up fade-up-${i+1}">
          <div class="flex items-center gap-2 mb-2">
            <div class="w-7 h-7 rounded-lg ${cd.bg} flex items-center justify-center">
              <i data-lucide="${cd.icon}" class="w-4 h-4 ${cd.color}"></i>
            </div>
            <span class="text-xs text-gray-400">${cd.label}</span>
          </div>
          <div class="text-lg sm:text-xl font-bold text-white">${cd.value}</div>
        </div>
      `).join('');
    }

    function renderOverallCharts() {
      const ops = getOps(allData);

      const revBars = [];
      let maxRev = 0;
      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        const total = m.aqR + m.reRe;
        revBars.push({ label: op, value: total });
        maxRev = Math.max(maxRev, total);
      });

      document.getElementById('chart-revenue').innerHTML = revBars.map(b => {
        const h = maxRev > 0 ? (b.value / maxRev) * 100 : 0;
        return `
          <div class="flex-1 flex flex-col items-center gap-1">
            <span class="text-[10px] text-gray-300 font-medium">${fmt(b.value)}</span>
            <div class="w-full rounded-sm bg-emerald-500" style="height:${Math.max(h,3)}%"></div>
            <span class="text-[8px] text-gray-500 truncate w-full text-center">${b.label}</span>
          </div>
        `;
      }).join('');

      const sesBars = [];
      let maxSes = 0;
      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        sesBars.push({ label: op, value: m.reSes });
        maxSes = Math.max(maxSes, m.reSes);
      });

      document.getElementById('chart-sessions').innerHTML = sesBars.map(b => {
        const h = maxSes > 0 ? (b.value / maxSes) * 100 : 0;
        return `
          <div class="flex-1 flex flex-col items-center gap-1">
            <span class="text-[10px] text-gray-300 font-medium">${fmtN(b.value)}</span>
            <div class="w-full rounded-sm bg-cyan-500" style="height:${Math.max(h,3)}%"></div>
            <span class="text-[8px] text-gray-500 truncate w-full text-center">${b.label}</span>
          </div>
        `;
      }).join('');

      const roasBars = [];
      let maxRoas = 0;
      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        roasBars.push({ label: op, value: m.roas });
        maxRoas = Math.max(maxRoas, m.roas);
      });

      document.getElementById('chart-roas').innerHTML = roasBars.map(b => {
        const h = maxRoas > 0 ? (b.value / maxRoas) * 100 : 0;
        return `
          <div class="flex-1 flex flex-col items-center gap-1">
            <span class="text-[10px] text-gray-300 font-medium">${b.value.toFixed(1)}x</span>
            <div class="w-full rounded-sm ${b.value >= 1 ? 'bg-amber-500' : 'bg-red-500'}" style="height:${Math.max(h,3)}%"></div>
            <span class="text-[8px] text-gray-500 truncate w-full text-center">${b.label}</span>
          </div>
        `;
      }).join('');

      let totalAQ = 0, totalRE = 0, totalSes = 0, totalGasto = 0, totalRev = 0;
      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        totalAQ += m.aqR;
        totalRE += m.reRe;
        totalSes += m.reSes;
        totalGasto += m.aqG;
        totalRev += m.totalRev;
      });

      document.getElementById('rev-aq').textContent = fmt(totalAQ);
      document.getElementById('rev-re').textContent = fmt(totalRE);
      document.getElementById('ses-total').textContent = fmtN(totalSes);
      document.getElementById('roas-spent').textContent = fmt(totalGasto);
      document.getElementById('roas-value').textContent = totalGasto > 0 ? (totalRev / totalGasto).toFixed(2) + 'x' : '-';
    }

    function renderOpsGrid() {
      const ops = getOps(allData);
      const grid = document.getElementById('ops-grid');

      grid.innerHTML = ops.map(op => {
        const m = getMetricsForOp(allData, op);
        const roasAQ = m.aqG > 0 ? m.aqR / m.aqG : 0;
        const roasInv = m.aqG > 0 ? (m.aqR + m.reRe) / m.aqG : 0;
        const totalRec = m.aqR + m.reRe;

        const totalForPie = Math.max(m.aqR + m.reRe, 1);
        const aqPercent = (m.aqR / totalForPie * 100).toFixed(1);
        const rePercent = (m.reRe / totalForPie * 100).toFixed(1);

        const pieRadius = 40;
        const circumference = 2 * Math.PI * pieRadius;
        const aqArc = (aqPercent / 100) * circumference;
        const reArc = (rePercent / 100) * circumference;

        const pieHTML = `
          <div class="flex items-center justify-center">
            <svg width="120" height="120" class="transform -rotate-90">
              <circle cx="60" cy="60" r="${pieRadius}" fill="none" stroke="#1a1a2e" stroke-width="16"/>
              <circle cx="60" cy="60" r="${pieRadius}" fill="none" stroke="#22c55e" stroke-width="16"
                      stroke-dasharray="${aqArc} ${circumference}" stroke-linecap="round"/>
              <circle cx="60" cy="60" r="${pieRadius}" fill="none" stroke="#6366f1" stroke-width="16"
                      stroke-dasharray="${reArc} ${circumference}" stroke-dashoffset="${-aqArc}" stroke-linecap="round"/>
            </svg>
            <div class="absolute text-center">
              <div class="text-sm font-bold text-white">${fmt(totalRec)}</div>
              <div class="text-xs text-gray-400">Total</div>
            </div>
          </div>
          <div class="text-xs space-y-1 mt-3">
            <div class="flex items-center justify-between">
              <div class="flex items-center gap-2"><div class="w-3 h-3 rounded-full bg-green-500"></div><span class="text-gray-400">AQ</span></div>
              <span class="text-white font-medium">${fmt(m.aqR)} (${aqPercent}%)</span>
            </div>
            <div class="flex items-center justify-between">
              <div class="flex items-center gap-2"><div class="w-3 h-3 rounded-full bg-indigo-500"></div><span class="text-gray-400">RE</span></div>
              <span class="text-white font-medium">${fmt(m.reRe)} (${rePercent}%)</span>
            </div>
          </div>
        `;

        const maxRet = Math.max(m.reRB, m.reRA, m.reRP, 1);
        const rbPercent = (m.reRB / maxRet * 100).toFixed(0);
        const raPercent = (m.reRA / maxRet * 100).toFixed(0);
        const rpPercent = (m.reRP / maxRet * 100).toFixed(0);

        const barsHTML = `
          <div class="space-y-2">
            <div>
              <div class="flex items-center justify-between mb-1">
                <span class="text-xs text-gray-400">B</span>
                <span class="text-xs font-medium text-white">${fmt(m.reRB)}</span>
              </div>
              <div class="w-full bg-gray-800 rounded h-4 overflow-hidden">
                <div class="h-full bg-blue-600" style="width: ${rbPercent}%"></div>
              </div>
            </div>
            <div>
              <div class="flex items-center justify-between mb-1">
                <span class="text-xs text-gray-400">A</span>
                <span class="text-xs font-medium text-white">${fmt(m.reRA)}</span>
              </div>
              <div class="w-full bg-gray-800 rounded h-4 overflow-hidden">
                <div class="h-full bg-amber-600" style="width: ${raPercent}%"></div>
              </div>
            </div>
            <div>
              <div class="flex items-center justify-between mb-1">
                <span class="text-xs text-gray-400">P</span>
                <span class="text-xs font-medium text-white">${fmt(m.reRP)}</span>
              </div>
              <div class="w-full bg-gray-800 rounded h-4 overflow-hidden">
                <div class="h-full bg-cyan-600" style="width: ${rpPercent}%"></div>
              </div>
            </div>
          </div>
        `;

        const sessionsHTML = `
          <div class="space-y-1 text-xs">
            <div class="flex justify-between"><span class="text-gray-400">B</span><span class="text-white font-medium">${fmtN(m.reSB)}</span></div>
            <div class="flex justify-between"><span class="text-gray-400">A</span><span class="text-white font-medium">${fmtN(m.reSA)}</span></div>
            <div class="flex justify-between"><span class="text-gray-400">P</span><span class="text-white font-medium">${fmtN(m.reSP)}</span></div>
            <div class="border-t border-gray-700 pt-1 mt-1 flex justify-between font-semibold"><span>Total</span><span class="text-cyan-400">${fmtN(m.reSes)}</span></div>
          </div>
        `;

        return `
          <div class="card-glass rounded-xl p-5 fade-up fade-up-1">
            <div class="flex items-center justify-between mb-5 pb-4 border-b border-white/10">
              <h3 class="text-lg font-bold text-white">${op}</h3>
              <div class="text-right">
                <div class="text-xs text-gray-400">ROAS Geral</div>
                <div class="text-2xl font-bold ${m.roas >= 1 ? 'text-emerald-400' : 'text-red-400'}">${m.roas.toFixed(2)}x</div>
              </div>
            </div>

            <div class="grid grid-cols-2 md:grid-cols-4 gap-3 mb-5">
              <div class="bg-white/5 rounded-lg p-3">
                <div class="text-xs text-gray-400 mb-1">Gasto Aquisição</div>
                <div class="text-lg font-bold text-red-400">${fmt(m.aqG)}</div>
              </div>
              <div class="bg-white/5 rounded-lg p-3">
                <div class="text-xs text-gray-400 mb-1">Receita Aquisição</div>
                <div class="text-lg font-bold text-emerald-400">${fmt(m.aqR)}</div>
              </div>
              <div class="bg-white/5 rounded-lg p-3">
                <div class="text-xs text-gray-400 mb-1">ROAS AQ</div>
                <div class="text-lg font-bold ${roasAQ >= 1 ? 'text-emerald-400' : 'text-red-400'}">${roasAQ.toFixed(2)}x</div>
              </div>
              <div class="bg-white/5 rounded-lg p-3">
                <div class="text-xs text-gray-400 mb-1">ROAS Investimento</div>
                <div class="text-lg font-bold ${roasInv >= 1 ? 'text-emerald-400' : 'text-red-400'}">${roasInv.toFixed(2)}x</div>
              </div>
            </div>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-5">
              <div>
                <h4 class="text-xs font-semibold text-gray-300 mb-4">Receita (AQ + RE)</h4>
                <div class="relative flex justify-center">${pieHTML}</div>
              </div>
              <div>
                <h4 class="text-xs font-semibold text-gray-300 mb-4">Receita Retenção (B/A/P)</h4>
                ${barsHTML}
              </div>
              <div>
                <h4 class="text-xs font-semibold text-gray-300 mb-4">Sessões (B/A/P)</h4>
                ${sessionsHTML}
              </div>
            </div>
          </div>
        `;
      }).join('');
    }

    function renderAll(data, updatedAt) {
      allData = data || [];
      const hasData = allData.length > 0;

      document.getElementById('empty-state').classList.toggle('hidden', hasData);
      document.getElementById('dashboard').classList.toggle('hidden', !hasData);

      document.getElementById('record-count').textContent = allData.length + ' registros';
      document.getElementById('last-update').textContent = updatedAt ? 'Atualizado em: ' + fmtDateTime(updatedAt) : '-';

      if (!hasData) {
        lucide.createIcons();
        return;
      }

      renderSummary();
      renderOverallCharts();
      renderOpsGrid();
      lucide.createIcons();
    }

    function loadData(showToast = false) {
      google.script.run
        .withSuccessHandler(function(result) {
          console.log('[dashboard] resposta syncData:', result);
          if (!result || typeof result !== 'object') {
            console.warn('syncData retornou resultado inválido:', result);
            renderAll([], null);
            return;
          }

          if (result.success === false) {
            console.error('syncData falhou:', result.error || 'Erro desconhecido');
            alert('Erro ao carregar dados: ' + (result.error || 'Falha na sincronização.'));
            renderAll([], result.updatedAt);
            return;
          }

          appConfig = result.config || appConfig;
          document.getElementById('main-title').textContent = appConfig.dashboard_title || 'Dashboard de Operações';
          renderAll(result.data || [], result.updatedAt);

          if (showToast) {
            const toast = document.getElementById('sync-toast');
            toast.classList.remove('hidden');
            setTimeout(() => toast.classList.add('hidden'), 2500);
          }
        })
        .withFailureHandler(function(err) {
          console.error(err);
          alert('Erro ao carregar dados: ' + (err.message || err));
        })
        .syncData();
    }

    document.getElementById('btn-sync').addEventListener('click', function() {
      const btn = document.getElementById('btn-sync');
      btn.disabled = true;
      loadData(true);
      setTimeout(function() {
        btn.disabled = false;
      }, 600);
    });

    loadData();
    lucide.createIcons();
  </script>
