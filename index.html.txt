  <script>
    let allData = [];
    let appConfig = {
      dashboard_title: 'Dashboard de Operações'
    };

    function fmt(v) {
      return 'R$ ' + Number(v || 0).toLocaleString('pt-BR', {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2
      });
    }

    function fmtN(v) {
      return Number(v || 0).toLocaleString('pt-BR');
    }

    function fmtUSD(v) {
      return '$' + Number(v || 0).toLocaleString('pt-BR', {
        minimumFractionDigits: 1,
        maximumFractionDigits: 1
      });
    }

    function fmtRoaPct(roas) {
      const pct = (Number(roas || 0) - 1) * 100;
      return pct.toFixed(1).replace('.', ',') + '%';
    }

    function fmtSignedBRL(v) {
      const n = Number(v || 0);
      if (n < 0) return '- ' + fmt(Math.abs(n));
      return fmt(n);
    }

    function fmtDateTime(v) {
      const d = new Date(v);
      return d.toLocaleString('pt-BR');
    }

    function getOps(data) {
      return [...new Set(data.map(d => d.operacao))].sort();
    }

    function getVal(data, op, dados) {
      const r = data.find(d => d.operacao === op && d.dados === dados);
      return r ? Number(r.valor || 0) : 0;
    }

    function getMetricsForOp(data, op) {
      const aqG = getVal(data, op, 'AQ - Gasto');
      const aqR = getVal(data, op, 'AQ - Receita');
      const reRB = getVal(data, op, 'RE - Receita - B');
      const reRA = getVal(data, op, 'RE - Receita - A');
      const reRP = getVal(data, op, 'RE - Receita - P');
      const reRe = reRB + reRA + reRP;
      const reSB = getVal(data, op, 'RE - Sessões - B');
      const reSA = getVal(data, op, 'RE - Sessões - A');
      const reSP = getVal(data, op, 'RE - Sessões - P');
      const reSes = reSB + reSA + reSP;
      const totalRev = aqR + reRe;
      const roas = aqG > 0 ? totalRev / aqG : 0;

      return { aqG, aqR, reRB, reRA, reRP, reRe, reSB, reSA, reSP, reSes, totalRev, roas };
    }

    function renderSummary() {
      const ops = getOps(allData);
      let totalGasto = 0, totalReceitaAQ = 0, totalReceitaRE = 0, totalSessoes = 0;

      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        totalGasto += m.aqG;
        totalReceitaAQ += m.aqR;
        totalReceitaRE += m.reRe;
        totalSessoes += m.reSes;
      });

      const totalReceitaTotal = totalReceitaAQ + totalReceitaRE;
      const roasGeral = totalGasto > 0 ? totalReceitaTotal / totalGasto : 0;

      const cards = [
        { label: 'Gasto Total', value: fmt(totalGasto), icon: 'trending-down', color: 'text-red-400', bg: 'bg-red-500/10' },
        { label: 'Receita Total', value: fmt(totalReceitaTotal), icon: 'trending-up', color: 'text-emerald-400', bg: 'bg-emerald-500/10' },
        { label: 'Sessões', value: fmtN(totalSessoes), icon: 'users', color: 'text-cyan-400', bg: 'bg-cyan-500/10' },
        { label: 'ROAS Geral', value: fmtRoaPct(roasGeral), icon: 'target', color: 'text-amber-400', bg: 'bg-amber-500/10' }
      ];

      document.getElementById('summary-cards').innerHTML = cards.map((cd, i) => `
        <div class="card-glass rounded-xl p-3 sm:p-4 fade-up fade-up-${i+1}">
          <div class="flex items-center gap-2 mb-2">
            <div class="w-7 h-7 rounded-lg ${cd.bg} flex items-center justify-center">
              <i data-lucide="${cd.icon}" class="w-4 h-4 ${cd.color}"></i>
            </div>
            <span class="text-xs text-gray-400">${cd.label}</span>
          </div>
          <div class="text-lg sm:text-xl font-bold text-white">${cd.value}</div>
        </div>
      `).join('');
    }

    function renderOverallCharts() {
      const ops = getOps(allData);

      const pieColors = ['#22c55e', '#6366f1', '#06b6d4', '#f59e0b', '#ef4444', '#8b5cf6', '#14b8a6', '#eab308', '#3b82f6', '#ec4899'];
      const spendData = ops.map((op, idx) => {
        const m = getMetricsForOp(allData, op);
        return { label: op, value: m.aqG, color: pieColors[idx % pieColors.length] };
      }).filter(item => item.value > 0);

      const totalSpend = spendData.reduce((acc, cur) => acc + cur.value, 0);
      const pieRadius = 44;
      const circumference = 2 * Math.PI * pieRadius;
      let offset = 0;

      const pieSlices = spendData.map(item => {
        const pct = totalSpend > 0 ? (item.value / totalSpend) : 0;
        const arc = pct * circumference;
        const slice = `
          <circle cx="56" cy="56" r="${pieRadius}" fill="none" stroke="${item.color}" stroke-width="14"
            stroke-dasharray="${arc} ${circumference}" stroke-dashoffset="${-offset}" stroke-linecap="round"></circle>
        `;
        offset += arc;
        return slice;
      }).join('');

      const tableRows = spendData.map(item => {
        const pct = totalSpend > 0 ? (item.value / totalSpend) * 100 : 0;
        return `
          <div class="grid grid-cols-[10px_1fr_auto_auto] items-center gap-2 text-[10px] py-0.5 border-b border-white/5">
            <span class="w-2.5 h-2.5 rounded-full" style="background:${item.color}"></span>
            <span class="text-gray-300 truncate">${item.label}</span>
            <span class="text-gray-200 whitespace-nowrap">${fmt(item.value)}</span>
            <span class="text-gray-400 whitespace-nowrap">${pct.toFixed(1)}%</span>
          </div>
        `;
      }).join('');

      document.getElementById('chart-revenue').innerHTML = `
        <div class="h-full flex items-center justify-between gap-2">
          <div class="w-[112px] shrink-0 flex items-center justify-center relative">
            <svg width="112" height="112" class="transform -rotate-90">
              <circle cx="56" cy="56" r="${pieRadius}" fill="none" stroke="#1a1a2e" stroke-width="14"></circle>
              ${pieSlices}
            </svg>
          </div>
          <div class="text-right pr-1">
            <div class="text-[10px] text-gray-400">Operações</div>
            <div class="text-2xl font-bold text-indigo-300">${ops.length}</div>
          </div>
        </div>
      `;
      document.getElementById('revenue-legend').innerHTML = `
        <div class="grid grid-cols-[1fr_auto_auto] gap-2 text-[10px] text-gray-500 pb-1 border-b border-white/10">
          <span>Operação</span><span>Gasto</span><span>%</span>
        </div>
        <div class="max-h-[98px] overflow-y-auto pr-1">
          ${tableRows || '<div class="text-[10px] text-gray-500 py-2">Sem gasto registrado.</div>'}
        </div>
      `;

      const lucroData = ops.map(op => {
        const m = getMetricsForOp(allData, op);
        return { label: op, value: m.totalRev - m.aqG };
      });
      const maxAbsLucro = Math.max(1, ...lucroData.map(i => Math.abs(i.value)));

      document.getElementById('chart-sessions').innerHTML = `
        <div class="h-full flex items-end gap-2">
          ${lucroData.map((item, idx) => {
            const magnitude = (Math.abs(item.value) / maxAbsLucro) * 42;
            const isPositive = item.value >= 0;
            return `
              <div class="flex-1 min-w-0 h-full flex flex-col items-center justify-center">
                <div class="text-[9px] ${isPositive ? 'text-emerald-300' : 'text-red-300'} whitespace-nowrap mb-1">${fmtSignedBRL(item.value)}</div>
                <div class="relative w-full h-20">
                  <div class="absolute left-0 right-0 top-1/2 border-t border-white/15"></div>
                  <div class="absolute left-1/2 -translate-x-1/2 w-6 ${isPositive ? 'bg-emerald-500' : 'bg-red-500'} rounded-sm"
                    style="${isPositive ? `bottom:50%;height:${Math.max(magnitude, 3)}%` : `top:50%;height:${Math.max(magnitude, 3)}%`}"></div>
                </div>
                <div class="text-[8px] text-gray-500 truncate w-full text-center">${item.label}</div>
              </div>
            `;
          }).join('')}
        </div>
      `;
      document.getElementById('lucro-legend').innerHTML = `
        <div class="grid grid-cols-[1fr_auto] gap-2 text-[10px] text-gray-500 pb-1 border-b border-white/10">
          <span>Operação</span><span>Lucro</span>
        </div>
        <div class="max-h-[98px] overflow-y-auto pr-1">
          ${lucroData.map(item => `
            <div class="grid grid-cols-[1fr_auto] gap-2 text-[10px] py-0.5 border-b border-white/5">
              <span class="text-gray-300 truncate">${item.label}</span>
              <span class="${item.value >= 0 ? 'text-emerald-300' : 'text-red-300'} whitespace-nowrap">${fmtSignedBRL(item.value)}</span>
            </div>
          `).join('')}
        </div>
      `;

      const roasBars = [];
      let maxRoas = 0;
      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        roasBars.push({ label: op, value: m.roas });
        maxRoas = Math.max(maxRoas, m.roas);
      });

      document.getElementById('chart-roas').innerHTML = roasBars.map(b => {
        const h = maxRoas > 0 ? (b.value / maxRoas) * 100 : 0;
        return `
          <div class="flex-1 flex flex-col items-center gap-1">
            <span class="text-[10px] text-gray-300 font-medium">${fmtRoaPct(b.value)}</span>
            <div class="w-full rounded-sm ${b.value >= 1 ? 'bg-amber-500' : 'bg-red-500'}" style="height:${Math.max(h,3)}%"></div>
            <span class="text-[8px] text-gray-500 truncate w-full text-center">${b.label}</span>
          </div>
        `;
      }).join('');

      let totalGasto = 0, totalRev = 0;
      ops.forEach(op => {
        const m = getMetricsForOp(allData, op);
        totalGasto += m.aqG;
        totalRev += m.totalRev;
      });
      document.getElementById('roas-spent').textContent = fmt(totalGasto);
      document.getElementById('roas-value').textContent = totalGasto > 0 ? fmtRoaPct(totalRev / totalGasto) : '-';
    }

    function renderOpsGrid() {
      const ops = getOps(allData);
      const grid = document.getElementById('ops-grid');

      grid.innerHTML = ops.map(op => {
        const m = getMetricsForOp(allData, op);
        const roasAQ = m.aqG > 0 ? m.aqR / m.aqG : 0;
        const roasInv = m.aqG > 0 ? (m.aqR + m.reRe) / m.aqG : 0;
        const totalRec = m.aqR + m.reRe;

        const totalForPie = Math.max(m.aqR + m.reRe, 1);
        const aqPercent = (m.aqR / totalForPie * 100).toFixed(1);
        const rePercent = (m.reRe / totalForPie * 100).toFixed(1);

        const pieRadius = 65;
        const circumference = 2 * Math.PI * pieRadius;
        const aqArc = (aqPercent / 100) * circumference;
        const reArc = (rePercent / 100) * circumference;
        const totalRecLabel = fmt(totalRec);
        const centerTextClass = totalRecLabel.length > 16
          ? 'text-[9px] sm:text-[11px]'
          : totalRecLabel.length > 13
            ? 'text-[10px] sm:text-xs'
            : 'text-sm sm:text-base';

        const pieHTML = `
          <div class="flex flex-col sm:flex-row items-center sm:items-start justify-center gap-4">
            <div class="relative flex items-center justify-center shrink-0">
              <svg width="196" height="196" class="transform -rotate-90">
                <circle cx="98" cy="98" r="${pieRadius}" fill="none" stroke="#1a1a2e" stroke-width="20"/>
                <circle cx="98" cy="98" r="${pieRadius}" fill="none" stroke="#22c55e" stroke-width="20"
                      stroke-dasharray="${aqArc} ${circumference}" stroke-linecap="round"/>
                <circle cx="98" cy="98" r="${pieRadius}" fill="none" stroke="#6366f1" stroke-width="20"
                      stroke-dasharray="${reArc} ${circumference}" stroke-dashoffset="${-aqArc}" stroke-linecap="round"/>
              </svg>
              <div class="absolute text-center px-4 max-w-[130px]">
                <div class="${centerTextClass} font-bold text-white leading-tight whitespace-nowrap">${totalRecLabel}</div>
                <div class="text-xs text-gray-400">Total</div>
              </div>
            </div>
            <div class="text-xs space-y-2 min-w-[140px] sm:pt-3">
              <div class="flex items-center justify-between gap-3">
                <div class="flex items-center gap-2"><div class="w-3 h-3 rounded-full bg-green-500"></div><span class="text-gray-400">Aquisição</span></div>
                <span class="text-white font-medium whitespace-nowrap">${fmt(m.aqR)} (${aqPercent}%)</span>
              </div>
              <div class="flex items-center justify-between gap-3">
                <div class="flex items-center gap-2"><div class="w-3 h-3 rounded-full bg-indigo-500"></div><span class="text-gray-400">Retenção</span></div>
                <span class="text-white font-medium whitespace-nowrap">${fmt(m.reRe)} (${rePercent}%)</span>
              </div>
            </div>
          </div>
        `;

        const totalRet = Math.max(m.reRB + m.reRA + m.reRP, 1);
        const rbPercent = (m.reRB / totalRet * 100);
        const raPercent = (m.reRA / totalRet * 100);
        const rpPercent = (m.reRP / totalRet * 100);

        function retentionBar(label, value, percent, colorClass) {
          const pct = percent.toFixed(1);
          const width = Math.max(percent, 14);
          return `
            <div>
              <div class="flex items-center justify-between mb-1">
                <span class="text-xs text-gray-400">${label}</span>
                <span class="text-xs font-medium text-white">${fmt(value)}</span>
              </div>
              <div class="w-full bg-gray-800 rounded h-6 overflow-hidden relative">
                <div class="h-full ${colorClass} flex items-center justify-end pr-2" style="width: ${width}%">
                  <span class="text-[10px] font-semibold text-white">${pct}%</span>
                </div>
              </div>
            </div>
          `;
        }

        const barsHTML = `
          <div class="space-y-2">
            ${retentionBar('Broadcast', m.reRB, rbPercent, 'bg-blue-600')}
            ${retentionBar('Automação', m.reRA, raPercent, 'bg-amber-600')}
            ${retentionBar('Push', m.reRP, rpPercent, 'bg-cyan-600')}
          </div>
        `;

        const maxSesBar = Math.max(m.reSB, m.reSA, m.reSP, 1);
        const rpsB = m.reSB > 0 ? (m.reRB / m.reSB) * 1000 : 0;
        const rpsA = m.reSA > 0 ? (m.reRA / m.reSA) * 1000 : 0;
        const rpsP = m.reSP > 0 ? (m.reRP / m.reSP) * 1000 : 0;
        const maxRpsBar = Math.max(rpsB, rpsA, rpsP, 1);
        const sesBars = [
          { label: 'Broadcast', sessoes: m.reSB, rps: rpsB, colorSessions: 'bg-blue-600', colorRps: 'bg-blue-300' },
          { label: 'Automação', sessoes: m.reSA, rps: rpsA, colorSessions: 'bg-amber-600', colorRps: 'bg-amber-300' },
          { label: 'Push', sessoes: m.reSP, rps: rpsP, colorSessions: 'bg-cyan-600', colorRps: 'bg-cyan-300' }
        ];

        const sessionsHTML = `
          <div>
            <div class="h-44 flex items-end justify-around gap-2 mb-1">
              ${sesBars.map(sb => {
                const heightSessions = ((sb.sessoes / maxSesBar) * 68) + 6;
                const heightRps = ((sb.rps / maxRpsBar) * 68) + 6;
                const labelBottomSessions = Math.min(heightSessions + 2, 84);
                const labelBottomRps = Math.min(heightRps + 2, 84);
                return `
                  <div class="flex-1 max-w-[112px] flex flex-col items-center gap-1">
                    <div class="text-[10px] text-gray-400 font-semibold">${sb.label}</div>
                    <div class="w-full h-30 flex items-end justify-center gap-2 px-1 pb-1">
                      <div class="w-9 flex flex-col items-center justify-end gap-1">
                        <div class="w-full h-24 relative flex items-end">
                          <span class="absolute left-1/2 -translate-x-1/2 text-[10px] text-gray-200 font-medium leading-none whitespace-nowrap" style="bottom: calc(${labelBottomSessions}% + 3px)">${fmtN(sb.sessoes)}</span>
                          <div class="w-full ${sb.colorSessions} rounded-t-sm" style="height:${Math.max(heightSessions, 4)}%"></div>
                        </div>
                        <span class="text-[10px] text-gray-400">Sessões</span>
                      </div>
                      <div class="w-9 flex flex-col items-center justify-end gap-1">
                        <div class="w-full h-24 relative flex items-end">
                          <span class="absolute left-1/2 -translate-x-1/2 text-[10px] text-gray-200 font-medium leading-none whitespace-nowrap" style="bottom: calc(${labelBottomRps}% + 3px)">${fmtUSD(sb.rps)}</span>
                          <div class="w-full ${sb.colorRps} rounded-t-sm" style="height:${Math.max(heightRps, 4)}%"></div>
                        </div>
                        <span class="text-[10px] text-gray-400">RPS</span>
                      </div>
                    </div>
                  </div>
                `;
              }).join('')}
            </div>
          </div>
        `;

        return `
          <div class="card-glass rounded-xl p-5 fade-up fade-up-1">
            <div class="flex items-center justify-between mb-5 pb-4 border-b border-white/10">
              <h3 class="text-lg font-bold text-white">${op}</h3>
              <div class="text-right">
                <div class="text-xs text-gray-400">ROAS Geral</div>
                <div class="text-xl font-bold ${m.totalRev - m.aqG >= 0 ? 'text-emerald-400' : 'text-red-400'} whitespace-nowrap">${fmtSignedBRL(m.totalRev - m.aqG)} / ${fmtRoaPct(m.roas)}</div>
              </div>
            </div>

            <div class="grid grid-cols-2 md:grid-cols-4 gap-3 mb-5">
              <div class="bg-white/5 rounded-lg p-3">

    function renderAll(data, updatedAt) {
      allData = data || [];
      const hasData = allData.length > 0;

      document.getElementById('empty-state').classList.toggle('hidden', hasData);
      document.getElementById('dashboard').classList.toggle('hidden', !hasData);

      document.getElementById('record-count').textContent = allData.length + ' registros';
      document.getElementById('last-update').textContent = updatedAt ? 'Atualizado em: ' + fmtDateTime(updatedAt) : '-';

      if (!hasData) {
        lucide.createIcons();
        return;
      }

      renderSummary();
      renderOverallCharts();
      renderOpsGrid();
      lucide.createIcons();
    }

    function loadData(showToast = false) {
      google.script.run
        .withSuccessHandler(function(rawResult) {
          let result = rawResult;
          if (typeof rawResult === 'string') {
            try {
              result = JSON.parse(rawResult);
            } catch (e) {
              console.error('Falha ao parsear syncDataJson:', e, rawResult);
              result = null;
            }
          }

          console.log('[dashboard] resposta syncData:', result);
          if (!result || typeof result !== 'object') {
            console.warn('syncData retornou resultado inválido:', result);
            renderAll([], null);
            return;
          }

          if (result.success === false) {
            console.error('syncData falhou:', result.error || 'Erro desconhecido');
            alert('Erro ao carregar dados: ' + (result.error || 'Falha na sincronização.'));
            renderAll([], result.updatedAt);
            return;
          }

          appConfig = result.config || appConfig;
          document.getElementById('main-title').textContent = appConfig.dashboard_title || 'Dashboard de Operações';
          renderAll(result.data || [], result.updatedAt);

          if (showToast) {
            const toast = document.getElementById('sync-toast');
            toast.classList.remove('hidden');
            setTimeout(() => toast.classList.add('hidden'), 2500);
          }
        })
        .withFailureHandler(function(err) {
          console.error(err);
          alert('Erro ao carregar dados: ' + (err.message || err));
        })
        .syncDataJson();
    }

    document.getElementById('btn-sync').addEventListener('click', function() {
      const btn = document.getElementById('btn-sync');
      btn.disabled = true;
      loadData(true);
      setTimeout(function() {
        btn.disabled = false;
      }, 600);
    });

    loadData();
    lucide.createIcons();
  </script>
